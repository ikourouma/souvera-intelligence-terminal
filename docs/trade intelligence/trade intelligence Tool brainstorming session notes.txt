## 2. The Data Engine: Calculating the Post-AGOA "Tariff Shock Risk"
To generate the exact numbers that Fortune 5 executives need, your backend must calculate the cost of losing trade preferences.
The following query identifies which alternative framework saves the most money if AGOA expires. It calculates the tariff penalty and ranks alternative markets based on financial efficiency.

WITH tariff_shock_calculation AS (
    -- Step 1: Isolate the baseline export asset values under threat
    SELECT 
        sp.country_iso3,
        sp.sector_key,
        sp.gross_export_usd,
        tt.mfn_rate AS us_mfn_rate,
        -- Calculate cash hit if forced to trade with USA under standard MFN rules
        (sp.gross_export_usd * (tt.mfn_rate / 100.0)) AS potential_annual_tariff_loss_usd
    FROM sector_performance sp
    JOIN treaty_tariffs tt ON tt.sector_key = sp.sector_key
    WHERE tt.treaty_id = 'US_AGOA'
),
alternative_treaty_lookup AS (
    -- Step 2: Query alternative active treaties for the country to find a backup market
    SELECT 
        tp.country_iso3,
        gt.treaty_id,
        gt.treaty_name,
        gt.sponsor_entity AS backup_market,
        alt_tt.sector_key,
        alt_tt.preference_rate AS alt_preference_tariff
    FROM treaty_participations tp
    JOIN global_treaties gt ON gt.treaty_id = tp.treaty_id
    JOIN treaty_tariffs alt_tt ON alt_tt.treaty_id = gt.treaty_id
    WHERE gt.treaty_id != 'US_AGOA' AND gt.is_active = TRUE
)-- Step 3: Combine datasets to output a clear risk-mitigation profileSELECT 
    tsc.country_iso3 AS country,
    tsc.sector_key AS sector,
    CONCAT('$', ROUND(tsc.gross_export_usd / 1000000, 2), 'M') AS active_exports,
    CONCAT('$', ROUND(tsc.potential_annual_tariff_loss_usd / 1000000, 2), 'M') AS us_tariff_shock_exposure,
    atl.treaty_name AS primary_backup_treaty,
    atl.backup_market AS replacement_destination_hub,
    CONCAT(atl.alt_preference_tariff, '%') AS backup_tariff_protection_rateFROM tariff_shock_calculation tscJOIN alternative_treaty_lookup atl ON atl.country_iso3 = tsc.country_iso3 AND atl.sector_key = tsc.sector_keyORDER BY tsc.potential_annual_tariff_loss_usd DESC;

import requestsimport jsonimport pandas as pdfrom typing import Dict, Any, List
class SouveraDataHarvester:
    """
    SOUVERA ENTERPRISE DATA INGESTION ENGINE
    Manages live API data orchestration from UN Comtrade, World Bank, and ITC.
    """
    def __init__(self, comtrade_key: str, itc_key: str):
        self.comtrade_key = comtrade_key
        self.itc_key = itc_key
        # Base production endpoints
        self.comtrade_url = "https://un.org"
        self.worldbank_url = "https://worldbank.org"

    def fetch_un_comtrade_flows(self, reporter_iso3: str, partner_iso3: str, hs_group: str, year: int = 2025) -> pd.DataFrame:
        """
        Extracts live asymmetric trade matrix flows directly from UN Comtrade v1.
        Filters out 'Re-exports' to eliminate the 'Rotterdam Effect' in Caribbean hubs.
        """
        headers = {"Ocp-Apim-Subscription-Key": self.comtrade_key}
        params = {
            "reporterCode": reporter_iso3,
            "partnerCode": partner_iso3,
            "cmdCode": hs_group, # e.g., '61,62,63' for Textiles & Apparel
            "flowCode": "M,X",    # Imports (M) and Exports (X)
            "period": str(year),
            "breakdownMode": "classic"
        }
        
        try:
            response = requests.get(self.comtrade_url, headers=headers, params=params, timeout=15)
            if response.status_code != 200:
                return pd.DataFrame() # Graceful degradation
                
            data = response.json().get('data', [])
            if not data:
                return pd.DataFrame()

            df = pd.DataFrame(data)
            # Retain critical attributes required for the Souvera Matrix
            refined_df = df[[
                'reporterCode', 'partnerCode', 'flowCode', 
                'cmdCode', 'primaryValue', 'isReexport'
            ]].copy()
            
            # Apply corporate risk filter: drop re-exports to preserve data purity
            refined_df = refined_df[refined_df['isReexport'] == False]
            return refined_df
            
        except requests.exceptions.RequestException:
            return pd.DataFrame()

    def fetch_world_bank_lpi(self, country_iso3: str) -> float:
        """
        Extracts the Logistics Performance Index (LPI) to use as a trade friction multiplier.
        """
        # World Bank Logistics Performance Index indicator code
        indicator = "LP.LPI.OVRL.RK" 
        url = f"{self.worldbank_url}/country/{country_iso3}/indicator/{indicator}?format=json&per_page=1"
        
        try:
            response = requests.get(url, timeout=10)
            data = response.json()
            if len(data) > 1 and data[1]:
                # Extract the most recent validated score (1-5 structural scale)
                latest_score = data[1][0].get('value')
                return float(latest_score) if latest_score else 2.5
            return 2.5
        except (requests.exceptions.RequestException, IndexError, ValueError):
            return 2.5 # Safe mid-tier baseline fallback

import unittestfrom decimal import Decimal
# =====================================================================# CORE ENGINE MODULE (The Logic Under Test)# =====================================================================
class SouveraTariffEngine:
    """
    SOUVERA CORE ENGINE
    Calculates exact tariff shock penalties and checks Rules of Origin compliance.
    """
    @staticmethod
    def calculate_rvc_percentage(export_value: Decimal, foreign_materials_value: Decimal) -> Decimal:
        """
        Calculates Regional Value Content using the Transaction Value Method.
        RVC % = ((Export Value - Foreign Materials Value) / Export Value) * 100
        """
        if export_value <= 0:
            return Decimal('0.00')
        rvc = ((export_value - foreign_materials_value) / export_value) * 100
        return rvc.quantize(Decimal('0.01'))

    @staticmethod
    def evaluate_treaty_eligibility(
        sector: str,
        export_value: Decimal,
        foreign_materials_value: Decimal,
        has_tariff_shift_occurred: bool,
        treaty_rules: dict
    ) -> dict:
        """
        Validates compliance against complex Rules of Origin, including De Minimis exceptions.
        """
        export_value = Decimal(str(export_value))
        foreign_materials_value = Decimal(str(foreign_materials_value))
        
        # Calculate base thresholds
        rvc_pct = SouveraTariffEngine.calculate_rvc_percentage(export_value, foreign_materials_value)
        foreign_ratio = (foreign_materials_value / export_value) * 100 if export_value > 0 else Decimal('100.00')
        
        # De Minimis Rule Exception evaluation
        passes_de_minimis = foreign_ratio <= Decimal(str(treaty_rules['de_minimis_cap_pct']))
        
        # Comprehensive Rule of Origin Check
        is_compliant = False
        if treaty_rules['rule_type'] == 'CTC_OR_RVC':
            # Passes if a major transformation happened OR local value content threshold is hit
            if has_tariff_shift_occurred or rvc_pct >= Decimal(str(treaty_rules['min_rvc_pct'])):
                is_compliant = True
        elif treaty_rules['rule_type'] == 'STRICT_CTC_AND_RVC':
            # Must satisfy both structural requirements simultaneously
            if has_tariff_shift_occurred and rvc_pct >= Decimal(str(treaty_rules['min_rvc_pct'])):
                is_compliant = True
                
        # If standard logic fails, check if saved by De Minimis allowance
        if not is_compliant and passes_de_minimis:
            is_compliant = True

        # Calculate Financial Exposure (Tariff Shock)
        # If compliant, applied rate is Preference Rate. If non-compliant, MFN rate applies.
        applied_rate = Decimal(str(treaty_rules['preference_rate'])) if is_compliant else Decimal(str(treaty_rules['mfn_rate']))
        tariff_shock_penalty = export_value * (Decimal(str(treaty_rules['mfn_rate'])) / 100) if not is_compliant else Decimal('0.00')

        return {
            "is_compliant": is_compliant,
            "rvc_percentage": float(rvc_pct),
            "applied_tariff_rate_pct": float(applied_rate),
            "tariff_shock_penalty_usd": float(tariff_shock_penalty.quantize(Decimal('0.01')))
        }
